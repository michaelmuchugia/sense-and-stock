AfriQuant — Phase 1: Database, Schema & Infrastructure Super Prompt
MISSION
Set up the AfriQuant production database infrastructure using Neon Postgres, Drizzle ORM, and Vercel deployment. By the end of this phase you will have a fully configured database with a complete schema, seeded with the existing mock data, a working connection from the Next.js app, and a secure foundation ready for the scraper in Phase 2.
This phase does not touch the UI. It does not touch the score engine logic. It does not touch any component. It is purely infrastructure and data layer.

SECURITY POSTURE — READ BEFORE WRITING ANYTHING
Security is not an afterthought. Apply these rules from the first line of code.
Environment variables

Every secret lives in .env.local for local development and in Vercel Environment Variables for production
Never hardcode connection strings, API keys, or secrets anywhere in the codebase
.env.local must be in .gitignore — verify this before the first commit
Use separate Neon database branches for development and production — Neon supports this natively

Database security

The Neon connection string used in the app must be the pooled connection string for serverless — not the direct connection
Direct connection string is used only for Drizzle migrations — never in application code
Enable SSL on all connections — Neon requires this by default, verify it is not being bypassed
Use Drizzle's prepared statements throughout — never string-interpolate values into queries
Row-level security will be added when auth is introduced — design the schema to support it from the start

API route security

The cron ingestion endpoint must be protected — Vercel provides a CRON_SECRET mechanism, implement it from day one
No database query runs in a public API route without validation
All environment variables must be validated at startup using a schema validation approach

What must never happen

No .env file committed to git ever
No raw SQL string concatenation
No direct database connection string in any component or client-side code
No unprotected cron endpoints

  const required = [
    'DATABASE_URL',
    'CRON_SECRET',
  ]

  const missing = required.filter(key => !process.env[key])

  if (missing.length > 0) {
    throw new Error(
      `Missing required environment variables: ${missing.join(', ')}\n` +
      `Check .env.local and Vercel Environment Variables.`
    )
  }
}

typescriptimport { db } from '../index'
import { stocks, stockPrices, stockScores } from '../schema'
import { eq, desc } from 'drizzle-orm'

export async function getAllStocks() {
  return db.select().from(stocks).where(eq(stocks.isActive, true))
}

export async function getStockByTicker(ticker: string) {
  const result = await db
    .select()
    .from(stocks)
    .where(eq(stocks.ticker, ticker.toUpperCase()))
    .limit(1)
  return result[0] ?? null
}

export async function getLatestPrice(stockId: number) {
  const result = await db
    .select()
    .from(stockPrices)
    .where(eq(stockPrices.stockId, stockId))
    .orderBy(desc(stockPrices.date))
    .limit(1)
  return result[0] ?? null
}

export async function upsertStock(data: {
  ticker: string
  name: string
  sector?: string
  profile?: string
}) {
  return db
    .insert(stocks)
    .values(data)
    .onConflictDoUpdate({
      target: stocks.ticker,
      set: {
        name: data.name,
        sector: data.sector,
        updatedAt: new Date(),
      },
    })
    .returning()
}

SEED SCRIPT
Migrate all mock data from lib/data/ into the database. This runs once on initial setup and can be re-run safely because all inserts use upsert logic.
scripts/seed.ts
typescriptimport * as dotenv from 'dotenv'
dotenv.config({ path: '.env.local' })

import { db } from '../lib/db/index'
import { STOCKS } from '../lib/data/stocks'
import { ANALYSTS } from '../lib/data/analysts'
import { SIG_SNAPSHOTS } from '../lib/data/snapshots'
import { upsertStock } from '../lib/db/queries/stocks'
import {
  stocks,
  analysts,
  analystTargets,
  analystCoverage,
  signalSnapshots,
  signalSnapshotEntries,
} from '../lib/db/schema'
import { eq } from 'drizzle-orm'

async function seed() {
  console.log('🌱 Starting AfriQuant database seed...')
  const startTime = Date.now()

  // ── 1. Seed stocks ────────────────────────────────────────────────
  console.log(`  Seeding ${STOCKS.length} stocks...`)
  for (const stock of STOCKS) {
    await upsertStock({
      ticker:   stock.s,
      name:     stock.n,
      sector:   stock.sec,
      profile:  stock.prof,
    })
  }
  console.log('  ✓ Stocks seeded')

  // ── 2. Seed analysts ──────────────────────────────────────────────
  console.log(`  Seeding ${ANALYSTS.length} analysts...`)
  for (const analyst of ANALYSTS) {
    await db
      .insert(analysts)
      .values({
        name:       analyst.nm,
        initials:   analyst.ini,
        firm:       analyst.firm,
        sinceYear:  parseInt(analyst.since),
        bio:        analyst.bio,
      })
      .onConflictDoUpdate({
        target: analysts.id,
        set: { name: analyst.nm },
      })
  }
  console.log('  ✓ Analysts seeded')

  // ── 3. Seed analyst coverage ──────────────────────────────────────
  console.log('  Seeding analyst coverage...')
  for (const analyst of ANALYSTS) {
    const dbAnalyst = await db
      .select()
      .from(analysts)
      .where(eq(analysts.name, analyst.nm))
      .limit(1)

    if (!dbAnalyst[0]) continue

    for (const ticker of analyst.stocks) {
      const dbStock = await db
        .select()
        .from(stocks)
        .where(eq(stocks.ticker, ticker))
        .limit(1)

      if (!dbStock[0]) continue

      await db
        .insert(analystCoverage)
        .values({
          analystId:  dbAnalyst[0].id,
          stockId:    dbStock[0].id,
          hitRatio:   analyst.hit.toString(),
          avgReturn:  analyst.ret.toString(),
          avgDays:    analyst.days,
          totalCalls: analyst.targets,
        })
        .onConflictDoUpdate({
          target: [analystCoverage.analystId, analystCoverage.stockId],
          set: {
            hitRatio:   analyst.hit.toString(),
            avgReturn:  analyst.ret.toString(),
            avgDays:    analyst.days,
            totalCalls: analyst.targets,
            updatedAt:  new Date(),
          },
        })
    }
  }
  console.log('  ✓ Analyst coverage seeded')

  // ── 4. Seed signal snapshots ──────────────────────────────────────
  console.log(`  Seeding ${SIG_SNAPSHOTS.length} signal snapshots...`)
  for (const snap of SIG_SNAPSHOTS) {
    const inserted = await db
      .insert(signalSnapshots)
      .values({
        month:    snap.month,
        label:    snap.label,
        naseRet:  snap.naseRet.toString(),
      })
      .onConflictDoUpdate({
        target: signalSnapshots.month,
        set: { naseRet: snap.naseRet.toString() },
      })
      .returning()

    if (!inserted[0]) continue

    for (let i = 0; i < snap.signals.length; i++) {
      const sig = snap.signals[i]
      const dbStock = await db
        .select()
        .from(stocks)
        .where(eq(stocks.ticker, sig.sym))
        .limit(1)

      if (!dbStock[0]) continue

      await db
        .insert(signalSnapshotEntries)
        .values({
          snapshotId:     inserted[0].id,
          stockId:        dbStock[0].id,
          entryPrice:     sig.entryPx.toString(),
          scoreAtSignal:  sig.score,
          note:           sig.note,
          rank:           i + 1,
        })
    }
  }
  console.log('  ✓ Signal snapshots seeded')

  const duration = Date.now() - startTime
  console.log(`\n✅ Seed complete in ${duration}ms`)
  process.exit(0)
}

seed().catch((err) => {
  console.error('❌ Seed failed:', err)
  process.exit(1)
})

CRON ENDPOINT — SECURED
Create the cron endpoint now even though the scraper logic comes in Phase 2. The security wrapper must be in place before any scraper code is written.
app/api/cron/ingest-prices/route.ts
typescriptimport { NextRequest, NextResponse } from 'next/server'
import { logIngestionRun } from '@/lib/db/queries/ingestion'

// Verify this request comes from Vercel Cron or an authorised caller
function authoriseCronRequest(request: NextRequest): boolean {
  const authHeader = request.headers.get('authorization')
  const cronSecret = process.env.CRON_SECRET

  if (!cronSecret) {
    console.error('CRON_SECRET environment variable is not set')
    return false
  }

  return authHeader === `Bearer ${cronSecret}`
}

export async function POST(request: NextRequest) {
  // Security check — first line, no exceptions
  if (!authoriseCronRequest(request)) {
    return NextResponse.json(
      { error: 'Unauthorised' },
      { status: 401 }
    )
  }

  const startTime = Date.now()

  try {
    // Scraper logic will be injected here in Phase 2
    // For now confirm the endpoint is reachable and secured
    console.log('AfriQuant ingestion job triggered')

    await logIngestionRun({
      source:         'nse_scraper',
      stocksUpdated:  0,
      status:         'success',
      durationMs:     Date.now() - startTime,
    })

    return NextResponse.json({
      success:        true,
      stocksUpdated:  0,
      durationMs:     Date.now() - startTime,
      message:        'Ingestion endpoint ready. Scraper logic added in Phase 2.',
    })

  } catch (error) {
    const message = error instanceof Error ? error.message : 'Unknown error'

    await logIngestionRun({
      source:         'nse_scraper',
      stocksUpdated:  0,
      status:         'failed',
      errorMessage:   message,
      durationMs:     Date.now() - startTime,
    })

    return NextResponse.json(
      { error: 'Ingestion failed', message },
      { status: 500 }
    )
  }
}
